async function handleFile(kind, file) {
  const token = Symbol();
  state.pendingFiles.set(kind, token);
  state[kind] = null;
  invalidate();
  $('analyze').disabled = true;
  try {
    if (file) {
      status(`正在本地读取 ${file.name}…`);
      const sheets = await readXlsx(file);
      if (state.pendingFiles.get(kind) !== token) return;
      state[kind] = sheets;
    }
    if (kind === 'counts') {
      state.countSpecs.clear();
      $('countsInfo').textContent = file ? `${file.name} · ${state.counts.length} 个表单` : '';
      sheetEditor();
      $('sheetDetails').open = [...state.countSpecs.values()].some(spec =>
        spec.active && (!spec.month || spec.columns['customer id'] == null || spec.columns.counts == null));
    } else {
      state.mapSpecs.clear();
      $('mapInfo').textContent = file ? file.name : '';
      const sheets = state.maps || [];
      $('mapSelectBox').classList.toggle('hidden', !sheets.length);
      $('mapSheet').innerHTML = sheets.map(([name]) => `<option value="${E(name)}">${E(name)}</option>`).join('');
      const candidates = sheets.map(([name, rows]) => {
        const spec = guessSpec(rows, FIELDS);
        return {name, score: FIELDS.slice(0, 4).filter(field => spec.columns[field] != null).length};
      }).sort((a, b) => b.score - a.score);
      if (candidates.length) $('mapSheet').value = candidates[0].name;
      $('mapSelectBox').open = !!candidates.length && candidates[0].score < 4;
      renderMappingFields();
    }
    status('已读取文件，可以生成对比。');
  } catch (error) {
    if (state.pendingFiles.get(kind) !== token) return;
    state[kind] = null;
    if (kind === 'counts') {
      state.countSpecs.clear();
      $('countsInfo').textContent = '读取失败';
      sheetEditor();
    } else {
      state.mapSpecs.clear();
      $('mapInfo').textContent = '读取失败';
      $('mapSelectBox').classList.add('hidden');
      $('mappingFields').innerHTML = '';
    }
    status(error.message, 'error');
  } finally {
    if (state.pendingFiles.get(kind) === token) state.pendingFiles.delete(kind);
    $('analyze').disabled = state.pendingFiles.size > 0;
  }
}

function importData() {
  if (!state.counts) throw Error('请先导入月度交易文件。');
  if ($('mapFile').files.length && !state.maps) throw Error('请重新导入客户映射文件，或清除该文件后查看整体数据。');
  const selected = [];
  const monthMap = Object.create(null);
  for (const [name, rows] of state.counts) {
    const spec = state.countSpecs.get(name);
    if (!spec?.active) continue;
    selected.push([name, projectRows(rows, spec, ['customer id', 'counts'], name)]);
    monthMap[name] = spec.month;
  }
  let mappingName = '空白映射';
  let maps = [[mappingName, [[1, FIELDS]]]];
  if (state.maps) {
    mappingName = $('mapSheet').value;
    const entry = state.maps.find(([name]) => name === mappingName);
    if (!entry) throw Error('请选择客户映射表单。');
    maps = [[mappingName, projectRows(entry[1], state.mapSpecs.get(mappingName), FIELDS, mappingName)]];
  }
  // Overall monthly sums are independent of how absent individual customers are treated.
  const allowZero = !state.maps || $('zeroConfirm').checked;
  return loadData(selected, maps, monthMap, mappingName, allowZero);
}

async function analyze() {
  if (state.pendingFiles.size) return;
  $('analyze').disabled = true;
  status('正在汇总交易笔数…');
  await new Promise(resolve => setTimeout(resolve, 0));
  try {
    const data = importData();
    state.data = data;
    state.monthIndex = new Map(data.months.map((period, index) => [period, index]));
    state.allYears = [...new Set(data.months.map(period => period.slice(0, 4)))];
    state.years = new Set(state.allYears);
    state.months = new Set(MONTHS);
    state.groups = state.maps ? aggregate(data.clients, 'group_key') : [];
    state.groupIndex = new Map(state.groups.map(group => [group.key, group]));
    state.clientIndex = new Map(data.clients.map(client => [client.key, client]));
    state.total = data.months.map((_, index) => safe(data.clients.reduce((sum, client) => sum + client.values[index], 0)));
    state.mode = 'yoy';
    state.target = data.months.at(-1);
    state.focus = Number(state.target.slice(-2));
    state.group = state.client = null;
    state.page = 0;
    $('detailSearch').value = '';
    $('momMonth').innerHTML = data.months.map(period => `<option>${period}</option>`).join('');
    $('momMonth').value = state.target;
    renderFilters();
    $('results').classList.remove('hidden');
    $('emptyState').classList.add('hidden');
    $('importPanel').open = false;
    render();
    status(`已导入 ${data.months.length} 个月的数据。`, 'success');
  } catch (error) {
    invalidate();
    status(error.message, 'error');
  } finally {
    $('analyze').disabled = state.pendingFiles.size > 0;
  }
}

function monthPeriod(year, calendarMonth) {
  return `${year}-${String(calendarMonth).padStart(2, '0')}`;

  return COLORS[state.allYears.indexOf(year) % COLORS.length];

function niceStep(raw) {
  if (raw <= 1) return 1;
  const order = 10 ** Math.floor(Math.log10(raw));
  const choice = [1, 2, 2.5, 5, 10].find(factor => factor * order >= raw);
  return Math.ceil(choice * order);
}

function chartAxis(values, zoom) {
  if (!values.length) return {min: 0, max: 1, step: 1};
  const smallest = Math.min(...values);
  const largest = Math.max(...values);
  const padding = Math.max((largest - smallest) * 0.12, largest * 0.005, 1);
  const lower = zoom ? Math.max(0, smallest - padding) : 0;
  const upper = largest + (zoom && lower > 0 ? padding : 0);
  const step = niceStep(Math.max(1, upper - lower) / 5);
  const min = Math.floor(lower / step) * step;
  return {min, max: Math.max(min + step, Math.ceil(upper / step) * step), step};
}

function barSvg(view, values, minWidth, zoom) {
  const points = view.groups.flatMap(group => group.points);
  const amounts = points.map(point => valueAt(values, point.period)).filter(value => value != null);
  const {min, max, step} = chartAxis(amounts, zoom);
  const left = Math.max(67, fmt(max).length * 7 + 19);
  const right = 15;
  const seriesCount = Math.max(...view.groups.map(group => group.points.length));
  const width = Math.max(minWidth, 540, left + right + view.groups.length * (seriesCount * 20 + 15));
  const groupWidth = (width - left - right) / view.groups.length;
  const pitch = Math.min(68, (groupWidth - 12) / seriesCount);
  const barWidth = Math.min(54, pitch - 4);
  const longest = Math.max(1, ...amounts.map(value => fmt(value).length));
  const rotate = longest * 7 > pitch - 3 && seriesCount > 1;
  const legend = view.groups[0].calendarMonth == null ? [] : view.groups[0].points;
  const legendColumns = Math.max(1, Math.floor((width - left - right) / 90));
  const legendHeight = Math.ceil(legend.length / legendColumns) * 20;
  const top = (rotate ? Math.max(45, longest * 7 + 17) : 38) + legendHeight;
  const height = Math.max(394, top + 255);
  const bottom = height - 45;
  const y = value => bottom - (value - min) / (max - min) * (bottom - top);
  const coord = number => number.toFixed(2);
  let svg = `<svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${height}" viewBox="0 0 ${width} ${height}" role="group" aria-label="交易笔数柱状图" style="font-family:system-ui,'Microsoft YaHei',sans-serif;background:white"><title>交易笔数对比</title><desc>纵轴从${fmt(min)}笔起。横轴按月份对齐。横线符号表示缺整月数据。</desc>`;
  legend.forEach((point, index) => {
    const x = left + index % legendColumns * 90;
    const row = Math.floor(index / legendColumns) * 20;
    svg += `<rect x="${x}" y="${row + 7}" width="10" height="10" rx="2" fill="${point.color}"/><text x="${x + 16}" y="${row + 16}" font-size="12" fill="#465d64">${E(point.label)}</text>`;
  });
  for (let value = min; value <= max; value += step) {
    const lineY = coord(y(value));
    svg += `<line x1="${left}" y1="${lineY}" x2="${width - right}" y2="${lineY}" stroke="${value === min ? '#9cb3b8' : '#e5eeef'}"/><text x="${left - 10}" y="${coord(y(value) + 4)}" text-anchor="end" fill="#5b7078" font-size="11">${fmt(value)}</text>`;
  }
  if (min > 0) svg += `<path d="M${left - 5},${bottom - 3} l4,4 l4,-4 l4,4" fill="none" stroke="#7b9397" stroke-width="1.5"/>`;
  view.groups.forEach((group, index) => {
    const center = left + (index + 0.5) * groupWidth;
    const totalWidth = group.points.length * pitch;
    group.points.forEach((point, seriesIndex) => {
      const centerX = center - totalWidth / 2 + (seriesIndex + 0.5) * pitch;
      const value = valueAt(values, point.period);
      const caption = `${point.period} ${value == null ? '缺整月数据' : `${fmt(value)} 笔`}`;
      if (value == null) {
        svg += `<text x="${coord(centerX)}" y="${bottom - 5}" text-anchor="middle" font-size="12" fill="#72868b"><title>${caption}</title>—</text>`;
        return;
      }
      const barTop = y(value);
      const barHeight = Math.max(0, bottom - barTop);
      if (barHeight === 0) {
        svg += `<circle cx="${coord(centerX)}" cy="${bottom}" r="3" fill="${point.color}"><title>${caption}</title></circle>`;
      } else {
        svg += `<rect class="bar" tabindex="0" aria-label="${caption}" x="${coord(centerX - barWidth / 2)}" y="${coord(barTop)}" width="${coord(barWidth)}" height="${coord(barHeight)}" rx="2" fill="${point.color}"><title>${caption}</title></rect>`;
      }
      const labelX = coord(centerX + (rotate ? 3 : 0));
      const labelY = coord(barTop - 7);
      const transform = rotate ? ` transform="rotate(-90 ${labelX} ${labelY})"` : '';
      svg += `<text x="${labelX}" y="${labelY}" text-anchor="${rotate ? 'start' : 'middle'}" font-size="11" font-weight="650" fill="${point.color}"${transform}>${fmt(value)}</text>`;
    });
    const active = group.calendarMonth === state.focus;
    const click = group.calendarMonth == null ? '' : ` class="month-hit" tabindex="0" role="button" aria-label="查看${group.label}明细" data-focus="${group.calendarMonth}"`;
    svg += `<g${click}><rect x="${coord(center - groupWidth / 2 + 3)}" y="${bottom + 9}" width="${coord(groupWidth - 6)}" height="25" rx="5" fill="${active ? '#e6f4ef' : 'transparent'}"/><text x="${coord(center)}" y="${bottom + 26}" text-anchor="middle" fill="${active ? '#086b62' : '#465d64'}" font-size="12" font-weight="${active ? 720 : 550}">${E(group.label)}</text></g>`;
  });
  return {svg: svg + '</svg>', min, max, missing: points.length - amounts.length};
}

function renderChart(view, scope) {
  $('chartHeading').textContent = state.mode === 'yoy' ? '各年份同月交易量' : '月度交易量环比';
  $('chartContext').textContent = state.mode === 'yoy'
    ? `${scope.name} · ${state.years.size} 个年份 · ${state.months.size} 个月份`
    : `${scope.name} · ${view.focus[0].period} 与 ${view.focus[1].period}`;
  const chart = barSvg(view, scope.values, $('chart').clientWidth, $('axisMode').value === 'focus');
  $('chart').innerHTML = chart.svg;
  $('chartNote').innerHTML = `纵轴从 <strong>${fmt(chart.min)}</strong> 笔起，最高刻度 ${fmt(chart.max)} 笔。${chart.missing ? ' — 表示缺整月数据。' : ''}`;
  const values = view.focus.map(point => {
    const value = valueAt(scope.values, point.period);
    return `<span class="pill"><i class="swatch" style="background:${point.color}"></i>${point.period} <strong>${value == null ? '缺数据' : `${fmt(value)} 笔`}</strong></span>`;
  });
  const [delta, changeRate] = difference(scope.values, view.focus);
  if (delta != null) {
    const from = view.focus.at(-2).period;
    const to = view.focus.at(-1).period;
    const growth = changeRate == null ? '' : `（${changeRate > 0 ? '+' : ''}${pct(changeRate)}）`;
    values.push(`<span class="pill">${to} 比 ${from} <strong>${sign(delta)} 笔${growth}</strong></span>`);
  }
  $('valueBadges').innerHTML = values.join('');
  $('chart').querySelectorAll('[data-focus]').forEach(element => {
    element.onclick = () => setFocus(Number(element.dataset.focus));
    element.onkeydown = event => {
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        setFocus(Number(element.dataset.focus));
      }
    };
  });
}

function difference(values, periods) {
  if (periods.length < 2) return [];
  const baseline = valueAt(values, periods.at(-2).period);
  const target = valueAt(values, periods.at(-1).period);
  return baseline == null || target == null ? [null, null] : [target - baseline, rate(baseline, target)];
}

function displayedObjects(scope, focus) {
  if (!scope.objects) return [];
  const groupRows = scope.level === '整体';
  const nameField = groupRows ? 'group name' : 'customer name';
  const idField = groupRows ? 'group id' : 'customer id';
  const query = $('detailSearch').value.trim().toLowerCase();
  const target = focus.at(-1).period;
  return scope.objects.map(item => ({item, name: item[nameField], id: item[idField] || item.key.replace(/^U:/, '')}))
    .filter(row => !query || `${row.name} ${row.id}`.toLowerCase().includes(query))
    .sort((a, b) => (valueAt(b.item.values, target) ?? -1) - (valueAt(a.item.values, target) ?? -1) || a.name.localeCompare(b.name));
}

function numberCell(value, kind = 'count') {
  if (value == null) return '<td class="num na">—</td>';
  const style = kind === 'count' ? '' : value > 0 ? ' up' : value < 0 ? ' down' : '';
  const shown = kind === 'rate' ? `${value > 0 ? '+' : ''}${pct(value)}` : kind === 'delta' ? sign(value) : fmt(value);
  return `<td class="num${style}">${shown}</td>`;
}

function detailValues(values, focus) {
  return focus.map(point => numberCell(valueAt(values, point.period))).join('') +
    difference(values, focus).map((value, index) => numberCell(value, index ? 'rate' : 'delta')).join('');
}

function renderDetails(view, scope) {
  const drill = !!scope.objects?.length;
  $('focusControl').classList.toggle('hidden', state.mode !== 'yoy' || !drill);
  $('detailSearch').classList.toggle('hidden', !drill);
  $('pager').classList.add('hidden');

    $('detailHeading').textContent = 'Client 各月笔数';
    $('detailHint').textContent = `${scope.name} · ${scope.client.key}`;
    const heads = state.mode === 'yoy' ? view.groups[0].points.map(point => point.label) : ['笔数'];
    $('detailTable').innerHTML = `<div class="tablewrap"><table><thead><tr><th>月份</th>${heads.map(label => `<th class="num">${E(label)}</th>`).join('')}</tr></thead><tbody>${view.groups.map(group => `<tr><td>${group.label}</td>${group.points.map(point => numberCell(valueAt(scope.values, point.period))).join('')}</tr>`).join('')}</tbody></table></div>`;
    $('detailNote').textContent = '';
    return;
  }
  if (!drill) {
    $('detailHeading').textContent = 'Group 与 Client';
    $('detailHint').textContent = '';
    $('detailTable').innerHTML = '<p class="hint">导入客户映射文件并重新生成，即可查看 Group 和 Client。</p>';
    $('detailNote').textContent = '';
    return;
  }
  $('detailHeading').textContent = scope.group ? 'Client 明细' : 'Group 明细';
  $('detailHint').textContent = scope.group ? '点击 Client 名称查看各月柱状图。' : '点击 Group 名称查看该集团的柱状图和 Client。';
  $('focusMonth').innerHTML = view.groups.filter(group => group.calendarMonth != null).map(group => `<option value="${group.calendarMonth}" ${group.calendarMonth === state.focus ? 'selected' : ''}>${group.label}</option>`).join('');
  const records = displayedObjects(scope, view.focus);
  const pageCount = Math.max(1, Math.ceil(records.length / PAGE_SIZE));
  state.page = Math.min(state.page, pageCount - 1);
  const shown = records.slice(state.page * PAGE_SIZE, (state.page + 1) * PAGE_SIZE);
  const hasDelta = view.focus.length >= 2;
  const tail = hasDelta ? '<th class="num">变化笔数</th><th class="num">变化率</th>' : '';
  const header = `<tr><th>${scope.group ? 'Client' : 'Group'}</th>${view.focus.map(point => `<th class="num">${point.period}</th>`).join('')}${tail}</tr>`;
  const total = `<tr class="totals"><td>${E(scope.name)}合计</td>${detailValues(scope.values, view.focus)}</tr>`;
  const rows = shown.map((row, index) => `<tr><td><button class="link-btn" data-drill="${index}">${E(row.name)}<span class="object-id">${E(row.id)}</span></button></td>${detailValues(row.item.values, view.focus)}</tr>`).join('');
  $('detailTable').innerHTML = `<div class="tablewrap"><table><thead>${header}</thead><tbody>${total}${rows}</tbody></table></div>${!records.length ? '<p class="hint">没有匹配的明细。</p>' : ''}`;
  $('detailTable').querySelectorAll('[data-drill]').forEach(button => {
    button.onclick = () => {
      const item = shown[Number(button.dataset.drill)].item;
      if (scope.group) selectScope(scope.group.key, item.key);
      else selectScope(item.key);
    };
  });
  $('pager').classList.toggle('hidden', pageCount <= 1);
  $('pageNote').textContent = `${fmt(records.length)} 条 · 第 ${state.page + 1} / ${pageCount} 页`;
  $('prevPage').disabled = state.page === 0;
  $('nextPage').disabled = state.page + 1 === pageCount;
  $('detailNote').textContent = hasDelta ? `变化比较 ${view.focus.at(-1).period} 与 ${view.focus.at(-2).period}。— 表示数据缺失或变化率无法计算。` : '';
}

}

function exportSheets(view, scope) {
  const periods = view.groups.flatMap(group => group.points);
  const chartRows = [['范围', '期间', '笔数', '整月数据已导入'], ...periods.map(point => [
    scope.name, point.period, valueAt(scope.values, point.period), state.monthIndex.has(point.period) ? '是' : '否',
  ])];
  const result = [['柱状图数据', chartRows]];
  const objects = displayedObjects(scope, view.focus);
  if (scope.objects?.length) {
    const rows = [['名称', 'ID', ...periods.map(point => point.period)]];
    rows.push([`${scope.name}合计`, '', ...periods.map(point => valueAt(scope.values, point.period))]);
    for (const row of objects) rows.push([row.name, row.id, ...periods.map(point => valueAt(row.item.values, point.period))]);
    result.push([scope.group ? 'Client明细' : 'Group明细', rows]);
  }
  if (view.focus.length >= 2) {
    const comparison = [['名称', 'ID', ...view.focus.map(point => point.period), '变化笔数', '变化率']];
    comparison.push([`${scope.name}合计`, '', ...view.focus.map(point => valueAt(scope.values, point.period)), ...difference(scope.values, view.focus)]);
    for (const row of objects) comparison.push([row.name, row.id, ...view.focus.map(point => valueAt(row.item.values, point.period)), ...difference(row.item.values, view.focus)]);
    result.push(['重点月份比较', comparison]);
  }
  return result;
}

function changeChoices(event, selected, convert, onChange) {
  const input = event.target;
  if (input.type !== 'checkbox') return;
  const key = convert(input.value);
  if (input.checked) selected.add(key);
  else if (selected.size > 1) selected.delete(key);
  else input.checked = true;
  onChange();
}
