
SELECT
    CUST AS CUSTOMER_ID,
    PO@CDT AS TXN_DATE,
    COUNT(*) AS ACH_COUNT
FROM YOUR_TABLE
WHERE PO@SID IS NOT NULL
  AND TRIM(PO@SID) <> ''
GROUP BY
    CUST,
    PO@CDT
ORDER BY
    CUST,
    PO@CDT

SELECT
    PO@CDT AS TXN_DATE,
    PO@SID AS BATCH_ID,
    COUNT(*) AS BATCH_COUNT
FROM YOUR_TABLE
WHERE PO@SID IS NOT NULL
GROUP BY
    PO@CDT,
    PO@SID
ORDER BY
    BATCH_COUNT DESC




SELECT
    PO@CDT AS TXN_DATE,
    INT(PO@CTM / 10000) AS TXN_HOUR,
    MOD(INT(PO@CTM / 100), 100) AS TXN_MINUTE,
    CASE
        WHEN PO@SID IS NULL
          OR TRIM(PO@SID) = ''
        THEN 'PP'
        ELSE 'ACH'
    END AS PAYMENT_TYPE,
    COALESCE(TRIM(PO@SID), '') AS BATCH_ID,
    COUNT(*) AS TXN_COUNT
FROM YOUR_TABLE
GROUP BY
    PO@CDT,
    INT(PO@CTM / 10000),
    MOD(INT(PO@CTM / 100), 100),
    CASE
        WHEN PO@SID IS NULL
          OR TRIM(PO@SID) = ''
        THEN 'PP'
        ELSE 'ACH'
    END,
    COALESCE(TRIM(PO@SID), '')
ORDER BY
    TXN_DATE,
    TXN_HOUR,
    TXN_MINUTE
