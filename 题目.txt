
SELECT
    PO@CDT AS TXN_DATE,
    INT(PO@CTM / 10000) AS TXN_HOUR,
    MOD(INT(PO@CTM / 100), 100) AS TXN_MINUTE,
    COUNT(*) AS TXN_COUNT
FROM YOUR_TABLE
GROUP BY
    PO@CDT,
    INT(PO@CTM / 10000),
    MOD(INT(PO@CTM / 100), 100)
ORDER BY
    TXN_DATE,
    TXN_HOUR,
    TXN_MINUTE

SELECT
    PO@CDT AS TXN_DATE,
    PO@SID AS BATCH_ID,
    COUNT(*) AS BATCH_COUNT
FROM YOUR_TABLE
WHERE PO@SID IS NOT NULL
GROUP BY
    PO@CDT,
    PO@SID
ORDER BY
    BATCH_COUNT DESC
