
WITH DAILY_COUNT AS (
    SELECT
        DATE(PO@CDT) AS TXN_DATE,
        HOUR(PO@CDT) AS TXN_HOUR,
        MINUTE(PO@CDT) AS TXN_MINUTE,
        COUNT(*) AS TXN_COUNT
    FROM YOUR_TABLE
    GROUP BY
        DATE(PO@CDT),
        HOUR(PO@CDT),
        MINUTE(PO@CDT)
),
RANKED_COUNT AS (
    SELECT
        TXN_DATE,
        TXN_HOUR,
        TXN_MINUTE,
        TXN_COUNT,
        ROW_NUMBER() OVER (
            PARTITION BY TXN_HOUR, TXN_MINUTE
            ORDER BY TXN_COUNT DESC, TXN_DATE DESC
        ) AS RN
    FROM DAILY_COUNT
)
SELECT
    TXN_HOUR,
    TXN_MINUTE,
    TXN_COUNT AS MAX_TXN_COUNT,
    TXN_DATE AS PEAK_DATE
FROM RANKED_COUNT
WHERE RN = 1
ORDER BY
    TXN_HOUR,
    TXN_MINUTE
