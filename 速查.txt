  <script>
    // 可直接在这里修改初始编号和定价。
    // code = 高亮编号；detail = 下方定价或条件。
    const pricing = [
      {
        name: "Waiver", label: "豁免",
        items: [
          { code: "CNPBSCHA", detail: "—" },
          { code: "2", detail: "<50k" },
          { code: "14", detail: "ACH" }
        ]
      },
      {
        name: "Flat", label: "固定费率",
        items: [
          { code: "8", detail: "1.2" },
          { code: "9", detail: "1" },
          { code: "10", detail: "0.5" },
          { code: "17", detail: "0.8" }
        ]
      },
      {
        name: "Cap", label: "封顶",
        items: [
          { code: "1", detail: "5" },
          { code: "11", detail: "20" },
          { code: "12", detail: "15" },
          { code: "13", detail: "10" }
        ]
      },
      {
        name: "Discount", label: "折扣",
        items: [
          { code: "3", detail: "20%" },
          { code: "4", detail: "40%" },
          { code: "5", detail: "50%" },
          { code: "18", detail: "25%" }
        ]
      },
      {
        name: "组合", label: "组合定价",
        items: [
          { code: "15", detail: "1.2 · 20%" },
          { code: "16", detail: "1.2 · 50%" },
          { code: "31", detail: "1 · 标准" }
        ]
      }
    ];
    const storageKey = "pricing-quick-reference-v1";
    const sheet = document.getElementById("sheet");
    const groups = document.getElementById("groups");
    const editButton = document.getElementById("editButton");
    const status = document.getElementById("status");
    let editing = false;
    // 读取当前浏览器之前保存的编号。
    try {
      const saved = JSON.parse(localStorage.getItem(storageKey));
      if (Array.isArray(saved)) {
        pricing.forEach((group, g) => {
          group.items.forEach((item, i) => {
            const value = saved[g]?.[i];
            if (typeof value === "string" && value.trim()) {
              item.code = value;
            }
          });
        });
      }
    } catch {
      // 浏览器禁用存储时，仍可正常查看和修改。
    }
    function element(tag, className, text) {
      const node = document.createElement(tag);
      if (className) node.className = className;
      if (text !== undefined) node.textContent = text;
      return node;
    }
    pricing.forEach(group => {
      const section = element("section", "group");
      section.setAttribute("aria-label", group.name);
      const category = element("div", "category");
      category.append(
        element("h2", "", group.name),
        element("span", "", `${group.label} · ${group.items.length}项`)
      );
      const items = element("div", "items");
      group.items.forEach(item => {
        const cell = element("div", "item");
        const code = element("span", "code", item.code);
        code.classList.toggle("long", item.code.length > 4);
        const editor = element("label", "editor");
        editor.append(element("span", "", "编号 / 代码"));
        const input = element("input");
        input.type = "text";
        input.value = item.code;
        input.autocomplete = "off";
        input.spellcheck = false;
        input.setAttribute(
          "aria-label",
          `${group.name}，定价 ${item.detail} 的编号`
        );
        if (/^\d+$/.test(item.code)) input.inputMode = "numeric";
        input.addEventListener("input", () => {
          item.code = input.value.trim();
          code.textContent = item.code || "—";
          code.classList.toggle("long", item.code.length > 4);
        });
        editor.append(input);
        cell.append(code, editor, element("div", "detail", item.detail));
        items.append(cell);
      });
      section.append(category, items);
      groups.append(section);
    });
    editButton.addEventListener("click", () => {
      // 完成前检查空编号。
      if (editing) {
        const empty = [...groups.querySelectorAll("input")]
          .find(input => !input.value.trim());
        if (empty) {
          status.textContent = "编号不能为空，请补充后再完成。";
          empty.focus();
          return;
        }
      }
      editing = !editing;
      sheet.classList.toggle("editing", editing);
      editButton.setAttribute("aria-pressed", String(editing));
      editButton.textContent = editing ? "完成修改" : "修改编号";
      if (editing) {
        status.textContent = "可直接修改各项编号。";
      } else {
        try {
          const codes = pricing.map(group =>
            group.items.map(item => item.code)
          );
          localStorage.setItem(storageKey, JSON.stringify(codes));
          status.textContent = "编号已保存在当前浏览器。";
        } catch {
          status.textContent = "编号已更新；当前浏览器无法保存修改。";
        }
      }
    });
  </script>
